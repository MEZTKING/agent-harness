from roman import to_roman

for n, want in [(1, "I"), (4, "IV"), (9, "IX"), (14, "XIV"), (40, "XL"), (90, "XC"), (400, "CD"),
                (1994, "MCMXCIV"), (3999, "MMMCMXCIX")]:
    assert to_roman(n) == want, (n, to_roman(n))
for bad in (0, 4000, -3):
    try:
        to_roman(bad)
    except ValueError:
        continue
    raise SystemExit(f"to_roman({bad}) did not raise ValueError")
print("ok")
