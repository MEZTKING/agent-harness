PAIRS = [(1000, "M"), (900, "CM"), (500, "D"), (400, "CD"), (100, "C"), (90, "XC"), (50, "L"), (40, "XL"),
         (10, "X"), (9, "IX"), (5, "V"), (4, "IV"), (1, "I")]


def to_roman(n):
    """Roman numerals for 1 to 3999."""
    if not 1 <= n <= 3999:
        raise ValueError("1 to 3999 only")
    out = ""
    for value, letters in PAIRS:
        while n >= value:
            out += letters
            n -= value
    return out
