from pages import page

items = list(range(1, 11))
assert page(items, 1, 3) == [1, 2, 3], page(items, 1, 3)
assert page(items, 4, 3) == [10]
assert page(items, 5, 3) == []
try:
    page(items, 0, 3)
except ValueError:
    pass
else:
    raise SystemExit("page 0 did not raise ValueError")
print("ok")
